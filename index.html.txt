/* ------------------ 简单 hash（用于示例性地存储密码） ------------------ */
async function sha256hex(str){
  const enc = new TextEncoder().encode(str);
  const buf = await crypto.subtle.digest('SHA-256', enc);
  const hex = Array.from(new Uint8Array(buf)).map(b=>b.toString(16).padStart(2,'0')).join('');
  return hex;
}

/* ------------------ 任务系统（简单） ------------------ */
function checkTasksAndGrant(user){
  // If streak reached thresholds, award
  let granted=[];
  if(user.streak && user.streak % 3 === 0 && !user._granted3){ // grant once per crossing
    user.incense = (user.incense||0)+1;
    user._granted3 = user.streak;
    user.logs.unshift({t: new Date().toISOString(), text:`连续拜 ${user.streak} 天，奖励 1 炷香`});
    granted.push('连续拜 3 天：+1 炷香');
  }
  if(user.streak && user.streak % 7 === 0 && !user._granted7){
    user.incenseValue = (user.incenseValue||0)+10;
    user._granted7 = user.streak;
    user.logs.unshift({t: new Date().toISOString(), text:`连续拜 ${user.streak} 天，奖励 10 香火值`});
    granted.push('连续拜 7 天：+10 香火值');
  }
  if(granted.length) saveUser(user);
  return granted;
}

/* ------------------ 拜香主流程 ------------------ */
function randomInt(min,max){ return Math.floor(Math.random()*(max-min+1))+min; }

function prayGod(user, godId){
  // returns result object
  if((user.incense||0) < 1) return {ok:false, msg:'香不够了'};
  // deduct
  user.incense = user.incense - 1;
  // increment lastPrayDate and streak
  const today = todayStr();
  if(user.lastPrayDate === today){
    // same day multiple prayers ok
  }else{
    // new day
    if(user.lastPrayDate){
      const diff = daysBetween(user.lastPrayDate, today);
      if(diff === 1) user.streak = (user.streak||0) + 1;
      else user.streak = 1;
    }else{
      user.streak = 1;
    }
    user.lastPrayDate = today;
  }

  // random reward: incenseValue +5~15
  const addXH = randomInt(5,15);
  user.incenseValue = (user.incenseValue||0) + addXH;

  // random small buff or resource
  const rolls = [
    {type:'buff', title:'今日学习效率 +10%', tip: randomStudyTip()},
    {type:'resource', title:'资源包：高频词汇 3 个', tip: randomVocabPack()},
    {type:'resource', title:'口语万能句：1 条', tip: randomSentence()},
    {type:'buff', title:'心态加成 +5%', tip:'深呼吸 3 次，专注力回升'}
  ];
  const chosen = rolls[randomInt(0, rolls.length-1)];

  // record
  const rec = {date:new Date().toISOString(), godId, incenseChange:-1, incenseValueChange:addXH, reward:chosen};
  user.records.unshift(rec);
  if(user.records.length>MAX_HISTORY) user.records.length=MAX_HISTORY;
  user.logs.unshift({t:new Date().toISOString(), text:`向 ${godId} 拜香，获得 香火值 +${addXH}（${chosen.title}）`});
  saveUser(user);
  // global log
  pushGlobalLog(`${user.name} 向 ${godId} 拜香，获得 +${addXH} 香火值`);
  // task checks
  const granted = checkTasksAndGrant(user);
  return {ok:true, rec, granted};
}

    </form>
  `);
  document.getElementById('auth-close').addEventListener('click', closeModal);
  document.getElementById('auth-form').addEventListener('submit', async (e)=>{
    e.preventDefault();
    const email = document.getElementById('auth-email').value.trim().toLowerCase();
    const name = document.getElementById('auth-name').value.trim() || email.split('@')[0];
    const pass = document.getElementById('auth-pass').value;
    if(!email || !pass){ alert('邮箱和密码是必填'); return; }
    const hash = await sha256hex(pass);
    const users = loadUsers();
    if(users[email]){
      // try login
      const res = authenticate(email, hash);
      if(!res.ok){ alert(res.msg); return; }
    }else{
      // register
      const r = registerUser(email, name, hash);
      if(!r.ok){ alert(r.msg); return; }
    }
    closeModal();
    renderAll();
  });
}

/* estimate daily count simple */
function estimateGodCount(godId){
  const key = 'baikaoshen_global_log';
  const arr = JSON.parse(localStorage.getItem(key) || '[]');
  // count occurrences of godId in recent logs
  let c=0;
  for(let i=0;i<Math.min(100,arr.length);i++){
    if(arr[i].text && arr[i].text.includes(godId)) c++;
    if(arr[i].text && arr[i].text.includes('拜香')) c++;
  }
  return Math.max(c, randomInt(5,60));
}
